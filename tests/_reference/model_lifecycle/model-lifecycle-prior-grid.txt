class GridCanvas:
    @model_validator(mode="after")
    def _sync_native_state(self):
        self._native_state.update(self.stride, self.shape, self.enabled, self.spacing)
        return self

    def __copy__(self):
        result = super().__copy__()
        result._native_state = copy(self._native_state)
        return result

    def model_copy(self, *, update=None, deep=False):
        result = super().model_copy(update=update, deep=deep)
        result._sync_native_state()
        return result
