class EventedModel:
    def __copy__(self):
        result = super().__copy__()
        result._change_state = copy(self._change_state)
        return result

    def __deepcopy__(self, memo=None):
        result = super().__deepcopy__(memo)
        result._primary_view = None
        return result

    @property
    def _delay_check_semaphore(self) -> int:
        return self._change_state.depth

    @_delay_check_semaphore.setter
    def _delay_check_semaphore(self, value: int) -> None:
        self._change_state.depth = value

class ComparisonDelayer:
    def __init__(self, target: EventedModel):
        self._target = target
